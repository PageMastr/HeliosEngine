# engine/editorui — L4 EditorUI (EDITOR_ONLY; 07 §1.3, §1.4, §4.4): the Dear ImGui docking shell of
# helios-editor over the ToolsFramework — main menu, dock layout, Documents / Inspector (property
# grid) / History / Output / Viewport panels, status bar, command palette, dark / light /
# high-contrast themes, DPI scaling, an ImGui renderer on the Helios RHI (bindless textures, vertex
# pulling, render-graph passes) and the in-house UI test harness core that helios-uitest drives
# (item table, ui.* input injection, captures, layout lints).
#
# The ImGui item hooks the harness uses (src/imgui_item_hooks.cpp) are compiled into tp_imgui by
# third_party/CMakeLists.txt; see that file and docs/adr/ADR-0.18-imgui-test-engine-licence.md.

include(HeliosShaders)

# The built-in theme token files are embedded as raw string literals (each file stays far below
# MSVC's 16 KiB limit per literal piece).
set(HELIOS_EDUI_THEMES dark light high_contrast)
set(HELIOS_EDUI_THEME_CPP ${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_themes.cpp)
set(helios_edui_themes_src "// Generated by engine/editorui/CMakeLists.txt from engine/editorui/themes/*.jsonc.\n")
string(APPEND helios_edui_themes_src "#include \"embedded_themes.h\"\n\nnamespace helios::edui::detail {\n")
foreach(theme ${HELIOS_EDUI_THEMES})
  set(theme_file ${CMAKE_CURRENT_SOURCE_DIR}/themes/${theme}.jsonc)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${theme_file})
  file(READ ${theme_file} theme_text)
  if(theme_text MATCHES "\\)helios_theme\"")
    message(FATAL_ERROR "${theme_file} contains the raw-string delimiter )helios_theme\"")
  endif()
  string(REPLACE "_" ";" theme_parts ${theme})
  set(theme_symbol "kTheme")
  foreach(part ${theme_parts})
    string(SUBSTRING ${part} 0 1 first)
    string(TOUPPER ${first} first)
    string(SUBSTRING ${part} 1 -1 rest)
    string(APPEND theme_symbol "${first}${rest}")
  endforeach()
  string(APPEND helios_edui_themes_src "extern const char* const ${theme_symbol};\n")
  string(APPEND helios_edui_themes_src "const char* const ${theme_symbol} = R\"helios_theme(${theme_text})helios_theme\";\n")
endforeach()
string(APPEND helios_edui_themes_src "} // namespace helios::edui::detail\n")
file(CONFIGURE OUTPUT ${HELIOS_EDUI_THEME_CPP} CONTENT "${helios_edui_themes_src}" @ONLY)

file(GLOB HELIOS_EDITORUI_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
# imgui_item_hooks.cpp belongs to tp_imgui (see the header comment).
list(REMOVE_ITEM HELIOS_EDITORUI_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/imgui_item_hooks.cpp)
helios_module(editorui
  SOURCES ${HELIOS_EDITORUI_SOURCES} ${HELIOS_EDUI_THEME_CPP}
  DEPS helios::toolsfw helios::rhi helios::render helios::core helios::reflect
  PRIVATE_DEPS helios::tp::imgui helios::tp::sdl3)
target_include_directories(helios_editorui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
helios_shaders(helios_editorui NAMESPACE helios_editorui_shaders
  FILES ${CMAKE_CURRENT_SOURCE_DIR}/shaders/imgui.slang
        ${CMAKE_CURRENT_SOURCE_DIR}/shaders/present.slang
        ${CMAKE_CURRENT_SOURCE_DIR}/shaders/viewport.slang)

# ---------------------------------------------------------------------------------------------
# Tests (doctest): themes and contrast, pseudo-localization, chords, the item table and lints on a
# headless ImGui context, the shell and property grid driven through injected input, and the
# renderer on the Null RHI. The GPU goldens (ED-15) run through helios-uitest (apps/tools).
# ---------------------------------------------------------------------------------------------
file(GLOB HELIOS_EDITORUI_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp)
helios_test(editorui_tests
  SOURCES ${HELIOS_EDITORUI_TEST_SOURCES}
  DEPS helios::editorui helios_toolsfw_samples helios::tp::imgui helios::render)
if(TARGET editorui_tests)
  target_include_directories(editorui_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_compile_definitions(editorui_tests PRIVATE
    HELIOS_EDUI_FIXTURE_DIR="${PROJECT_SOURCE_DIR}/engine/toolsfw/samples/project"
    HELIOS_EDUI_GOLDEN_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/golden"
    HELIOS_EDUI_THEME_DIR="${CMAKE_CURRENT_SOURCE_DIR}/themes"
    HELIOS_EDUI_TEST_WORK_DIR="${CMAKE_CURRENT_BINARY_DIR}/editorui_test_work")
  set_tests_properties(editorui_tests PROPERTIES TIMEOUT 600)
  # Suites "gpu*" create a real EditorHost: a Vulkan device (lavapipe in CI) and SDL video on the
  # offscreen driver, so no display server is needed. test_main.cpp keeps them out of the run above.
  add_test(NAME editorui_tests_gpu COMMAND editorui_tests --test-suite=gpu*)
  set_tests_properties(editorui_tests_gpu PROPERTIES LABELS gpu TIMEOUT 300 ENVIRONMENT "SDL_VIDEO_DRIVER=offscreen")
  if(HELIOS_SANITIZE AND NOT WIN32)
    # LeakSanitizer: keep lavapipe loaded, as helios_editor_smoke does (apps/editor).
    set_tests_properties(editorui_tests_gpu PROPERTIES ENVIRONMENT_MODIFICATION "VK_LOADER_DISABLE_DYNAMIC_LIBRARY_UNLOADING=set:1")
  endif()
endif()
